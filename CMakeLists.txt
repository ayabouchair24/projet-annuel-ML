cmake_minimum_required(VERSION 3.15...3.30)
project(PA_V2 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)

# Force compatibility policy for older external dependencies like OSQP
if(POLICY CMP0000)
    cmake_policy(SET CMP0000 NEW)
endif()
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)

# Automatically download and set up OSQP from GitHub
include(FetchContent)

FetchContent_Declare(
        osqp
        GIT_REPOSITORY https://github.com/osqp/osqp.git
        GIT_TAG        v0.6.3
)

# Turn off unneeded extras to make it build faster
set(OSQP_BUILD_UNITTESTS OFF CACHE BOOL "" FORCE)
set(OSQP_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)

FetchContent_MakeAvailable(osqp)

# Your library configuration
add_library(PA_V2 SHARED
        src/RBF.cpp
#        src/SVM.cpp
        src/MLP.cpp
        src/modele_lineaire.cpp
        src/SVM_updated.cpp
)

# Link your project to OSQP using the correct target name generated by its CMakeLists
target_link_libraries(PA_V2 PRIVATE osqpstatic)
target_include_directories(PA_V2 PRIVATE ${osqp_SOURCE_DIR}/include)
